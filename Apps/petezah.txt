    function titleCaseWord(word) {
      if (!word) {
        return "";
      }
      if (/^[A-Z0-9]+$/.test(word)) {
        return word;
      }
      if (/^[a-z0-9]+$/.test(word) && word.length <= 4 && /\d/.test(word)) {

      return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();

        .replace(/([a-z])([A-Z])/g, "$1 $2")
        .replace(/([A-Za-z])(\d)/g, "$1 $2")
        .replace(/(\d)([A-Za-z])/g, "$1 $2")
        .replace(/\s+/g, " ")
        .trim()
        .split(" ")
        .map(titleCaseWord)
        .join(" ");
    }

    function guessTags(file, title) {
      const haystack = (file + " " + title).toLowerCase();
      const found = [];

      for (const entry of CATEGORY_MAP) {
        if (entry.keys.some((key) => haystack.includes(key.toLowerCase()))) {
          for (const tag of entry.tags) {
            if (!found.includes(tag)) {
              found.push(tag);
            }
          }
        }
      }

      if (!found.length) {
        found.push("Arcade");
      }

      return found.slice(0, 3);
    }

    function createGame(file) {
      const title = humanizeFileName(file);
      const first = title.charAt(0).toUpperCase();
      const letter = LETTERS.includes(first) ? first : "#";
      return {
        file,
        title,
        letter,
        searchText: (title + " " + file).toLowerCase(),
        tags: guessTags(file, title)
      };
    }

    const games = uniqueFiles
      .map(createGame)
      .sort((a, b) => a.title.localeCompare(b.title, undefined, { numeric: true, sensitivity: "base" }));

    function buildGroups() {
      const groups = Object.fromEntries(LETTERS.map((letter) => [letter, []]));
      for (const game of games) {
        if (groups[game.letter]) {
          groups[game.letter].push(game);
        }
      }
      return groups;
    }

    const groups = buildGroups();
    let currentGameFile = "";

    function updatePlayerButtons() {
      const hasGame = Boolean(currentGameFile);
      document.getElementById("fullscreenGame").disabled = !hasGame;
      document.getElementById("openInTab").disabled = !hasGame;
      document.getElementById("reloadGame").disabled = !hasGame;
      document.getElementById("closeGame").disabled = !hasGame;
    }

    function renderAlphabetNav() {
      const nav = document.getElementById("alphabetNav");
      nav.innerHTML = "";

      for (const letter of LETTERS) {
        const btn = document.createElement("button");
        btn.type = "button";
        btn.className = "alpha-btn glass";
        if (groups[letter].length) {
          btn.classList.add("has-games");
        }
        btn.textContent = letter;
        btn.dataset.target = "section-" + letter;

        btn.addEventListener("click", () => {
          const section = document.getElementById(btn.dataset.target);
          if (section && !section.classList.contains("hidden")) {
            section.scrollIntoView({ behavior: "smooth", block: "start" });
          }
        });

        nav.appendChild(btn);
      }
    }

    function renderCatalog() {
      const catalog = document.getElementById("catalog");
      catalog.innerHTML = "";

      for (const letter of LETTERS) {
        const section = document.createElement("section");
        section.className = "section glass";
        section.id = "section-" + letter;

        const head = document.createElement("div");
        head.className = "section-head";
        head.innerHTML = "<h2>" + letter + "</h2><div class=\"count\">" + groups[letter].length + " games</div>";

        const grid = document.createElement("div");
        grid.className = "grid";

        for (const game of groups[letter]) {
          const btn = document.createElement("button");
          btn.type = "button";
          btn.className = "launch-btn glass";
          btn.dataset.search = game.searchText;
          btn.dataset.letter = letter;
          btn.dataset.file = game.file;

          const tags = game.tags.map((tag) => "<span class=\"tag\">" + tag + "</span>").join("");
          btn.innerHTML =
            "<div class=\"title-wrap\">" +
              "<strong>" + escapeHtml(game.title) + "</strong>" +
              "<small>" + escapeHtml(game.file) + "</small>" +
            "</div>" +
            "<div class=\"tags\">" + tags + "</div>";

          btn.addEventListener("click", () => openGame(game.file));
          grid.appendChild(btn);
        }

        section.appendChild(head);
        section.appendChild(grid);
        catalog.appendChild(section);
      }
    }

    function escapeHtml(value) {
      return value
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll("\"", "&quot;")
        .replaceAll("'", "&#39;");
    }

    function updateStats(visibleGames, visibleSections, query) {
      const stats = document.getElementById("stats");
      const searchState = query ? "Filtered" : "Showing";
      stats.innerHTML = [
        "<div class=\"stat glass\"><strong>" + games.length + "</strong> total games</div>",
        "<div class=\"stat glass\"><strong>" + visibleGames + "</strong> visible games</div>",
        "<div class=\"stat glass\"><strong>" + visibleSections + "</strong> active sections</div>",
        "<div class=\"stat glass\">" + searchState + " catalog</div>"
      ].join("");
    }

    function filterCatalog() {
      const query = document.getElementById("searchInput").value.trim().toLowerCase();
      const sections = Array.from(document.querySelectorAll(".section"));
      const alphaButtons = Array.from(document.querySelectorAll(".alpha-btn"));

      let visibleGames = 0;
      let visibleSections = 0;

      for (const section of sections) {
        const buttons = Array.from(section.querySelectorAll(".launch-btn"));
        let localVisible = 0;

        for (const button of buttons) {
          const match = !query || button.dataset.search.includes(query);
          button.classList.toggle("hidden", !match);
          if (match) {
            localVisible += 1;
            visibleGames += 1;
          }
        }

        section.classList.toggle("hidden", localVisible === 0);
        const count = section.querySelector(".count");
        if (count) {
          count.textContent = localVisible + " games";
        }
        if (localVisible > 0) {
          visibleSections += 1;
        }
      }

      for (const alpha of alphaButtons) {
        const target = document.getElementById(alpha.dataset.target);
        const visible = target && !target.classList.contains("hidden");
        alpha.classList.toggle("active", Boolean(query) && visible);
        alpha.style.opacity = visible ? "1" : "0.35";
      }

      updateStats(visibleGames, visibleSections, query);

      const catalog = document.getElementById("catalog");
      const existingEmpty = document.getElementById("emptyState");
      if (!visibleGames) {
        if (!existingEmpty) {
          const empty = document.createElement("div");
          empty.id = "emptyState";
          empty.className = "empty glass";
          empty.textContent = "No games match that search.";
          catalog.appendChild(empty);
        }
      } else if (existingEmpty) {
        existingEmpty.remove();
      }
    }

    function setPlayerState({ open, loaded, title, message }) {
      const playerShell = document.getElementById("playerShell");
      const playerTitle = document.getElementById("playerTitle");
      const playerPlaceholder = document.getElementById("playerPlaceholder");

      playerShell.classList.toggle("is-open", open);
      playerShell.classList.toggle("is-loaded", loaded);

      if (title) {
        playerTitle.textContent = title;
      }

      if (message) {
        playerPlaceholder.textContent = message;
      }

      updatePlayerButtons();
    }

    async function openGame(file) {
      const normalized = normalizeFileName(file);
      const encoded = encodeURIComponent(normalized);
      const playerShell = document.getElementById("playerShell");
      const playerTitle = document.getElementById("playerTitle");
      const gameFrame = document.getElementById("gameFrame");
      currentGameFile = file;

      setPlayerState({
        open: true,
        loaded: false,
        title: humanizeFileName(file),
        message: "Loading " + humanizeFileName(file) + "..."
      });
      playerShell.scrollIntoView({ behavior: "smooth", block: "start" });

      try {
        const response = await fetch(CDN_BASE + encoded + "?t=" + Date.now());
        if (!response.ok) {
          throw new Error("Request failed with status " + response.status);
        }

        const text = await response.text();
        gameFrame.srcdoc = text;
        playerTitle.textContent = humanizeFileName(file);
        setPlayerState({
          open: true,
          loaded: true,
          title: humanizeFileName(file),
          message: ""
        });
      } catch (error) {
        document.getElementById("gameFrame").srcdoc = "";
        setPlayerState({
          open: true,
          loaded: false,
          title: humanizeFileName(file),
          message: "Unable to open " + normalized + ". " + error.message
        });
      }
    }

    async function openGameInNewTab(file) {
      const normalized = normalizeFileName(file);
      const encoded = encodeURIComponent(normalized);

      try {
        const response = await fetch(CDN_BASE + encoded + "?t=" + Date.now());
        if (!response.ok) {
          throw new Error("Request failed with status " + response.status);
        }

        const text = await response.text();
        const newWin = window.open("about:blank", "_blank");
        if (!newWin) {
          throw new Error("Pop-up blocked");
        }

        newWin.document.open();
        newWin.document.write(text);
        newWin.document.close();
      } catch (error) {
        alert("Unable to open " + normalized + " in a new tab. " + error.message);
      }
    }

    function bindControls() {
      const searchInput = document.getElementById("searchInput");
      const clearButton = document.getElementById("clearSearch");
      const fullscreenButton = document.getElementById("fullscreenGame");
      const openInTabButton = document.getElementById("openInTab");
      const reloadButton = document.getElementById("reloadGame");
      const closeButton = document.getElementById("closeGame");
      const gameFrame = document.getElementById("gameFrame");
      const frameWrap = document.getElementById("frameWrap");

      searchInput.addEventListener("input", filterCatalog);
      clearButton.addEventListener("click", () => {
        searchInput.value = "";
        filterCatalog();
        searchInput.focus();
      });

      fullscreenButton.addEventListener("click", async () => {
        if (!currentGameFile) {
          return;
        }

        try {
          if (document.fullscreenElement === frameWrap) {
            await document.exitFullscreen();
            return;
          }
          await frameWrap.requestFullscreen();
        } catch (error) {
          alert("Unable to enter fullscreen. " + error.message);
        }
      });

      openInTabButton.addEventListener("click", () => {
        if (currentGameFile) {
          openGameInNewTab(currentGameFile);
        }
      });

      reloadButton.addEventListener("click", () => {
        if (currentGameFile) {
          openGame(currentGameFile);
        }
      });

      closeButton.addEventListener("click", () => {
        currentGameFile = "";
        gameFrame.srcdoc = "";
        setPlayerState({
          open: false,
          loaded: false,
          title: "Select a game from the catalog",
          message: "Pick any title from the alphabet catalog and it will open right here in the page."
        });
      });
    }

    renderAlphabetNav();
    renderCatalog();
    bindControls();
    filterCatalog();
    updatePlayerButtons();
  </script>
